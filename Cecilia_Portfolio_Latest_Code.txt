    const { useState, useEffect, useRef } = React;

    function App() {
      const [activeTab, setActiveTab] = useState('HOME');
      const [isModalOpen, setIsModalOpen] = useState(false);
      const navRef = useRef(null);
      const navButtonRefs = useRef({});
      const sectionVisibilityRef = useRef({});
      const [activePill, setActivePill] = useState({ left: 0, width: 0, opacity: 0 });

      const navItems = [
        { id: 'HOME', label: 'Home', sectionId: 'home' },
        { id: 'EDUCATION', label: 'Education', sectionId: 'education' },
        { id: 'EXPERIENCE', label: 'Experience', sectionId: 'experience' },
        { id: 'PROJECTS', label: 'Projects', sectionId: 'projects' },
        { id: 'SKILLS', label: 'Skills', sectionId: 'skills' },
      ];

      const updateActivePill = () => {
        const nav = navRef.current;
        const button = navButtonRefs.current[activeTab];
        if (!nav || !button) return;
        setActivePill({
          left: button.offsetLeft,
          width: button.offsetWidth,
          opacity: 1,
        });
      };

      useEffect(() => {
        updateActivePill();
        const observer = typeof ResizeObserver !== 'undefined'
          ? new ResizeObserver(updateActivePill)
          : null;
        if (observer && navRef.current) observer.observe(navRef.current);
        window.addEventListener('resize', updateActivePill);
        return () => {
          if (observer) observer.disconnect();
          window.removeEventListener('resize', updateActivePill);
        };
      }, [activeTab]);

      useEffect(() => {
        const sections = navItems
          .map((item) => document.getElementById(item.sectionId))
          .filter(Boolean);

        const observer = new IntersectionObserver((entries) => {
          entries.forEach((entry) => {
            sectionVisibilityRef.current[entry.target.id] = entry.isIntersecting
              ? entry.intersectionRatio
              : 0;
          });

          const visibleItem = navItems.reduce((best, item) => {
            const ratio = sectionVisibilityRef.current[item.sectionId] || 0;
            return ratio > best.ratio ? { id: item.id, ratio } : best;
          }, { id: activeTab, ratio: 0 });

          if (visibleItem.ratio > 0) {
            setActiveTab((current) => current === visibleItem.id ? current : visibleItem.id);
            const visibleSection = navItems.find((item) => item.id === visibleItem.id);
            const nextHash = visibleSection ? '#' + visibleSection.sectionId : '';
            if (nextHash && window.location.hash !== nextHash) {
              window.history.replaceState(null, '', nextHash);
            }
          }
        }, {
          rootMargin: '-18% 0px -52% 0px',
          threshold: [0, 0.1, 0.25, 0.5, 0.75],
        });

        sections.forEach((section) => observer.observe(section));

        const initialHash = window.location.hash.replace('#', '').toLowerCase();
        if (navItems.some((item) => item.sectionId === initialHash)) {
          window.requestAnimationFrame(() => {
            document.getElementById(initialHash)?.scrollIntoView({ behavior: 'auto', block: 'start' });
          });
        }

        return () => observer.disconnect();
      }, []);

      const handleNavigate = (item) => {
        const target = document.getElementById(item.sectionId);
        if (!target) return;
        setActiveTab(item.id);
        window.history.replaceState(null, '', '#' + item.sectionId);
        target.scrollIntoView({ behavior: 'smooth', block: 'start' });
      };

      return (
        <div className="min-h-screen flex flex-col justify-between relative">
          
          {/* 1. 顶部悬浮胶囊导航 */}
          <header className="fixed top-5 left-0 right-0 z-40 flex justify-center px-4">
            <div className="bg-white/90 backdrop-blur-xl border border-slate-200/80 rounded-full px-5 py-2.5 shadow-xl shadow-slate-100 flex items-center space-x-6">
              
              <div 
                className="font-bold text-xs tracking-wider uppercase text-slate-900 cursor-pointer flex items-center space-x-2"
                onClick={() => handleNavigate(navItems[0])}
              >
                <div className="w-6 h-6 rounded-full bg-slate-900 text-white flex items-center justify-center font-bold text-[10px]">胡</div>
                <span>胡敏雪 (CECILIA HU)</span>
              </div>

              <nav ref={navRef} aria-label="Portfolio sections" className="relative flex items-center space-x-1 sm:space-x-2 text-xs font-medium text-slate-600">
                <span
                  aria-hidden="true"
                  className="pointer-events-none absolute bottom-0 top-0 rounded-full bg-[#FF5500] shadow-md transition-all duration-[260ms] ease-out"
                  style={{ left: activePill.left, width: activePill.width, opacity: activePill.opacity }}
                />
                {navItems.map((item) => (
                  <button
                    key={item.id}
                    ref={(node) => { navButtonRefs.current[item.id] = node; }}
                    onClick={() => handleNavigate(item)}
                    aria-current={activeTab === item.id ? 'page' : undefined}
                    className={`relative z-10 px-3.5 py-1.5 rounded-full transition-colors duration-200 ${
                      activeTab === item.id
                        ? 'text-white font-bold'
                        : 'hover:text-slate-900 hover:bg-slate-100'
                    }`}
                  >
                    {item.label}
                  </button>
                ))}
              </nav>

              <ContactMenu />
            </div>
          </header>

          {/* 2. 单页连续浏览：Navbar 快捷跳转 + 自然纵向滚动 */}
          <main>
            <section id="home" data-section="HOME" className="portfolio-section min-h-screen pt-28 pb-16 flex items-center">
              <div className="w-full">
              <HomeLarryStyle 
                onOpenMatcher={() => setIsModalOpen(true)} 
              />
              </div>
            </section>

            <section id="education" data-section="EDUCATION" className="portfolio-section min-h-[calc(100vh-5rem)] py-20 flex items-center">
              <div className="w-full"><EducationSection /></div>
            </section>

            <section id="experience" data-section="EXPERIENCE" className="portfolio-section py-24">
              <ExperienceSection />
            </section>

            <section id="projects" data-section="PROJECTS" className="portfolio-section py-24">
              <ProjectsSection />
            </section>

            <section id="skills" data-section="SKILLS" className="portfolio-section min-h-[calc(100vh-5rem)] scroll-mt-0 pt-16 pb-12 flex items-center">
              <div className="w-full"><SkillsSection /></div>

    function ContactMenu() {
      const email = 'hmxdxyx@163.com';
      const wechatId = 'cecilia1205hmx';
      const [isOpen, setIsOpen] = useState(false);
      const [copiedItem, setCopiedItem] = useState(null);
      const menuRef = useRef(null);
      const copyTimerRef = useRef(null);

      useEffect(() => {
        const handleOutsideClick = (event) => {
          if (menuRef.current && !menuRef.current.contains(event.target)) setIsOpen(false);
        };
        const handleEscape = (event) => {
          if (event.key === 'Escape') setIsOpen(false);
        };
        document.addEventListener('mousedown', handleOutsideClick);
        document.addEventListener('keydown', handleEscape);
        return () => {
          document.removeEventListener('mousedown', handleOutsideClick);
          document.removeEventListener('keydown', handleEscape);
          if (copyTimerRef.current) window.clearTimeout(copyTimerRef.current);
        };
      }, []);

      const copyContact = async (value, item) => {
        try {
          await navigator.clipboard.writeText(value);
        } catch (error) {
          const input = document.createElement('textarea');
          input.value = value;
          input.style.position = 'fixed';
          input.style.opacity = '0';
          document.body.appendChild(input);
          input.select();
          document.execCommand('copy');
          document.body.removeChild(input);
        }
        setCopiedItem(item);
        if (copyTimerRef.current) window.clearTimeout(copyTimerRef.current);
        copyTimerRef.current = window.setTimeout(() => setCopiedItem(null), 1800);
      };

      return (
        <div ref={menuRef} className="relative">
          <button
            type="button"
            aria-expanded={isOpen}
            aria-controls="contact-popover"
            onClick={() => setIsOpen(!isOpen)}
            className="group flex items-center gap-1.5 rounded-full bg-[#FF5500] px-4 py-1.5 text-xs font-bold text-white shadow-[0_8px_20px_rgba(255,85,0,.22)] transition-all duration-200 hover:px-5 hover:bg-[#E94E00] hover:shadow-[0_10px_24px_rgba(255,85,0,.30)] focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-orange-300 focus-visible:ring-offset-2"
          >
            Let’s Talk
            <span className="transition-transform duration-200 group-hover:-translate-y-0.5 group-hover:translate-x-0.5" aria-hidden="true">↗</span>
          </button>

            <div id="contact-popover" role="dialog" aria-label="Contact Cecilia" className="absolute right-0 top-full z-50 mt-2.5 w-[300px] max-w-[calc(100vw-1.5rem)] rounded-3xl border border-slate-200/50 bg-white/[0.92] p-4 shadow-[0_20px_48px_rgba(15,23,42,.10)] backdrop-blur-md">
              <div className="absolute -top-1.5 right-7 h-3 w-3 rotate-45 border-l border-t border-slate-200/50 bg-white/[0.92]" />

                </div>
                <button type="button" onClick={() => setIsOpen(false)} aria-label="关闭联系卡片" className="flex h-7 w-7 items-center justify-center rounded-full bg-slate-100 text-[10px] text-slate-500 transition-colors hover:bg-slate-200">✕</button>

              <div className="mt-4 space-y-2">
                <div className="flex h-11 items-center gap-2.5 rounded-xl px-2.5 transition-colors duration-200 hover:bg-slate-50">
                  <span className="flex h-7 w-7 flex-none items-center justify-center rounded-lg border border-slate-200 text-[12px] text-slate-600" aria-hidden="true">✉</span>
                  <a href={'mailto:' + email} className="min-w-0 flex-1 truncate text-[11px] font-semibold text-slate-700 transition-colors hover:text-[#FF5500]">{email}</a>

                    {copiedItem === 'email' ? 'Copied! ✓' : 'Copy'}
